# DnsRecord

## Example Usage

```typescript
import { DnsRecord } from "@unkey/api/models/components";

let value: DnsRecord = {
  type: "CNAME",
  name: "api.acme.com",
  value: "a1b2c3d4e5f6g7h8.cname.unkey.com",
  ttl: 60,
  verified: false,
  note: "Create as DNS-only if your provider offers the choice.",
};
```
